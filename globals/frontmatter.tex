% Course title string. Defined ONCE here and reused by the title block, the PDF
% metadata, and the running header, so the empty-\sectionnum case is handled in
% exactly one place. (Previously the running header rebuilt this string itself
% and emitted a double space whenever \sectionnum was empty.)
\ifdefempty{\sectionnum}{%
  \newcommand{\coursetitle}{\coursenum~-- \coursename}
}{%
  \newcommand{\coursetitle}{\coursenum~\sectionnum~-- \coursename}
}

% format title on first page
\makeatletter
\newcommand{\@semester}{\semester}
\newcommand{\@brandmark}{\includegraphics[width=\brandmarkwidth, alt={{\institution}}]{../../globals/\brandmark}}

\renewcommand{\maketitle}{%
\begin{center}%
    \@brandmark \\
    \LARGE \@title \\
    \Large \coursetitle \\
    \Large \@semester
\end{center}
}
\makeatother

% article metadata --- author comes from \profname (globals/<inst>/instructor.tex)
% so the theme carries no hardcoded name across courses or institutions
\hypersetup{
    pdftitle   = {\coursetitle~Course Syllabus},
    pdfauthor  = {\profname}
}
\title{Course Syllabus}
\author{\profname}
\date{\semester}

% format header for pages 2+
\usepackage{fancyhdr}
\pagestyle{fancy}
% header
\lhead{\profname}
\chead{\coursetitle}
\rhead{\semester}
% footer
\lfoot{}
\cfoot{\thepage}
\rfoot{}
